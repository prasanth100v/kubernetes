# ============================================
# 📦 Kubernetes Storage Commands Cheat Sheet
# ============================================

# 🔍 LIST RESOURCES
list_commands:
  pv: "kubectl get pv"                     # 📦 List all PersistentVolumes
  pvc: "kubectl get pvc"                  # 📄 List all PersistentVolumeClaims
  sc: "kubectl get sc"                    # 🗂️ List all StorageClasses
  pvc_namespace: "kubectl get pvc -n <namespace>"   # 📁 PVCs in specific namespace

# 🔎 DESCRIBE RESOURCES (Detailed Info)
describe_commands:
  pv: "kubectl describe pv <pv-name>"     # 🔍 Detailed PV info
  pvc: "kubectl describe pvc <pvc-name>"  # 🔍 Detailed PVC info
  sc: "kubectl describe sc <sc-name>"     # 🔍 StorageClass info
  storageclass: "kubectl describe storageclass <storage-class-name>"  # 🔍 Full SC details

# 🚀 CREATE RESOURCES
create_commands:
  pvc: "kubectl apply -f pvc.yaml"        # 📄 Create PVC using YAML file

# ❌ DELETE RESOURCES
delete_commands:
  pvc: "kubectl delete pvc <pvc-name>"    # 🗑️ Delete PVC (default namespace)
  pvc_namespace: "kubectl delete pvc <pvc-name> -n <namespace>"  # 🗑️ Delete PVC in specific namespace
  pv: "kubectl delete pv <pv-name>"       # 🗑️ Delete PersistentVolume

# 📊 DEBUG / CHECK STORAGE INSIDE POD
debug_commands:
  disk_usage: "kubectl exec -it <pod-name> -- df -h"   # 📊 Check disk usage inside pod
  list_mount: "kubectl exec -it <pod-name> -- ls /mnt" # 📂 List mounted files (replace /mnt path)

# ============================================
# 💡 PRO TIPS
# ============================================
tips:
  - "✅ Always check PVC status (Bound/Pending)"
  - "⚠️ Pending PVC → Check StorageClass & provisioner"
  - "🔐 Use encrypted storage in production"
  - "📍 Use namespace flag (-n) for multi-tenant clusters"
